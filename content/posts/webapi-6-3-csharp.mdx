# 5. Archivo nuevo: `CreateWorkOrderFromMaintenancePlanHandler.cs`

Ruta:

`Features/WorkOrders/Commands/CreateWorkOrderFromMaintenancePlanHandler.cs`

Este será el Handler que coordina la creación de la orden desde el plan.

```csharp
using AcSoft.Application.Abstractions.Persistence;
using AcSoft.Application.Features.WorkOrders.DTOs;
using AcSoft.Domain.Entities;
using AcSoft.Domain.Enums;

namespace AcSoft.Application.Features.WorkOrders.Commands;

public sealed class CreateWorkOrderFromMaintenancePlanHandler
{
    private readonly IMaintenancePlanRepository _maintenancePlanRepository;
    private readonly IActivityRepository _activityRepository;
    private readonly IWorkOrderRepository _workOrderRepository;
    private readonly IUnitOfWork _unitOfWork;

    public CreateWorkOrderFromMaintenancePlanHandler(
        IMaintenancePlanRepository maintenancePlanRepository,
        IActivityRepository activityRepository,
        IWorkOrderRepository workOrderRepository,
        IUnitOfWork unitOfWork)
    {
        _maintenancePlanRepository = maintenancePlanRepository;
        _activityRepository = activityRepository;
        _workOrderRepository = workOrderRepository;
        _unitOfWork = unitOfWork;
    }

    public async Task<Guid> HandleAsync(
        CreateWorkOrderFromMaintenancePlanRequest request,
        CancellationToken cancellationToken)
    {
        // 1. Obtener el plan de mantenimiento.
        var maintenancePlan =
            await _maintenancePlanRepository.GetByIdWithActivitiesAsync(
                request.MaintenancePlanId,
                cancellationToken);

        if (maintenancePlan is null)
        {
            throw new InvalidOperationException(
                "El plan de mantenimiento no existe.");
        }

        // 2. Validar que el plan esté activo.
        if (maintenancePlan.Status != MaintenancePlanStatus.Active)
        {
            throw new InvalidOperationException(
                "Solo se pueden generar órdenes desde un plan activo.");
        }

        // 3. Generar el número de la orden.
        var workOrderNumber = $"OT-{DateTime.UtcNow:yyyyMMddHHmmss}";

        // 4. Crear la orden de trabajo.
        var workOrder = WorkOrder.Create(
            number: workOrderNumber,
            customerId: request.CustomerId,
            equipmentId: request.EquipmentId);

        // 5. Recorrer las actividades del plan.
        foreach (var planActivity in maintenancePlan.Activities
                     .OrderBy(x => x.Moment)
                     .ThenBy(x => x.Sequence))
        {
            // Consultar la actividad del catálogo con sus preguntas.
            var activity =
                await _activityRepository.GetByIdWithQuestionsAsync(
                    planActivity.ActivityId,
                    cancellationToken);

            if (activity is null)
            {
                throw new InvalidOperationException(
                    $"No se encontró la actividad del catálogo " +
                    $"con identificador {planActivity.ActivityId}.");
            }

            // Agregar la actividad a la orden.
            workOrder.AddActivity(
                activity: activity,
                moment: planActivity.Moment,
                sequence: planActivity.Sequence,
                isUnplanned: false);
        }

        // 6. Agregar la orden al repositorio.
        await _workOrderRepository.AddAsync(
            workOrder,
            cancellationToken);

        // 7. Guardar los cambios.
        await _unitOfWork.SaveChangesAsync(
            cancellationToken);

        // 8. Retornar el identificador de la orden.
        return workOrder.Id;
    }
}
```

Este código conserva tus convenciones actuales:

- `HandleAsync`
- `AddAsync`
- `SaveChangesAsync`
- `CancellationToken`

Y utiliza las mismas interfaces que ya tienes.

## 2. Configuración de MaintenancePlan

Podemos configurar la relación utilizando el campo privado `_activities`.

Crea el archivo:

`AcSoft.Infrastructure/Persistence/Configurations/MaintenancePlanConfiguration.cs`

```csharp
using AcSoft.Domain.Entities;
using Microsoft.EntityFrameworkCore;
using Microsoft.EntityFrameworkCore.Metadata.Builders;

namespace AcSoft.Infrastructure.Persistence.Configurations;

public sealed class MaintenancePlanConfiguration
    : IEntityTypeConfiguration<MaintenancePlan>
{
    public void Configure(EntityTypeBuilder<MaintenancePlan> builder)
    {
        builder.ToTable("MaintenancePlans");

        builder.HasKey(x => x.Id);

        builder.Property(x => x.Code)
            .HasMaxLength(30)
            .IsRequired();

        builder.HasIndex(x => x.Code)
            .IsUnique();

        builder.Property(x => x.Name)
            .HasMaxLength(150)
            .IsRequired();

        builder.Property(x => x.Description)
            .HasMaxLength(500);

        builder.Property(x => x.Status)
            .IsRequired();

        builder.HasMany(x => x.Activities)
            .WithOne()
            .HasForeignKey(x => x.MaintenancePlanId)
            .OnDelete(DeleteBehavior.Cascade);

        builder.Navigation(x => x.Activities)
            .HasField("_activities")
            .UsePropertyAccessMode(PropertyAccessMode.Field);
    }
}
```

### ¿Qué estamos configurando?

- `HasMany`: un plan puede tener muchas actividades.
- `WithOne()`: cada actividad pertenece a un solo plan, sin necesitar una propiedad de navegación inversa.
- `HasForeignKey`: `MaintenancePlanActivity.MaintenancePlanId` es la llave foránea.
- `OnDelete(Cascade)`: si se elimina un plan, EF Core elimina sus registros asociados.
- `HasField("_activities")`: EF Core utiliza la colección privada del agregado, sin modificarla desde fuera.
- `HasIndex(x => x.Code).IsUnique()`: evita códigos duplicados.

`AcSoft.Infrastructure/Persistence/Configurations/MaintenancePlanActivityConfiguration.cs`

```csharp
using AcSoft.Domain.Entities;
using Microsoft.EntityFrameworkCore;
using Microsoft.EntityFrameworkCore.Metadata.Builders;

namespace AcSoft.Infrastructure.Persistence.Configurations;

public sealed class MaintenancePlanActivityConfiguration
    : IEntityTypeConfiguration<MaintenancePlanActivity>
{
    public void Configure(EntityTypeBuilder<MaintenancePlanActivity> builder)
    {
        builder.ToTable("MaintenancePlanActivities");

        builder.HasKey(x => x.Id);

        builder.Property(x => x.MaintenancePlanId)
            .IsRequired();

        builder.Property(x => x.ActivityId)
            .IsRequired();

        builder.Property(x => x.Moment)
            .IsRequired();

        builder.Property(x => x.Sequence)
            .IsRequired();

        builder.HasIndex(x => new
        {
            x.MaintenancePlanId,
            x.Moment,
            x.Sequence
        })
        .IsUnique();
    }
}
```

El índice único refuerza en la base de datos la regla que ya tienes en `AddActivity`: no puede repetirse la secuencia para un mismo momento dentro del mismo plan.

Una precisión: todavía no configuramos la relación entre `MaintenancePlanActivity` y `Activity`, porque necesitamos revisar tu entidad `Activity` y decidir si agregaremos una propiedad de navegación o mantendremos la relación únicamente mediante `ActivityId`.
